// Tactile postcard v1 (15.10.2026): pressure → heartbeat. One SF15-600 strip, one coin motor.
// Ported from Hugbeat heartbeat_v4.ino (Arduino Uno) to the ESP32-C3 SuperMini; petting/purring left out.
//
// Wiring:  3V3 ── SF15 ──┬── GPIO1            GPIO5 ── 1 kΩ ── PN2222 base
//                        └── 10 kΩ ── GND      motor + ── 5V, motor − ── collector, emitter ── GND
//
// Behaviour: at start 1 s calibration (don't press). Pressing makes each beat stronger and slowly "charges"
// the card; the charge sets the tempo (55 → 140 bpm) and drains slowly after letting go. No press + no charge = still.
// Serial (115200): raw, pressure 0..1, charge 0..1, bpm, every 100 ms — for tuning the constants below.
#include <Arduino.h>

static const int PIN_FSR = 1, PIN_MOTOR = 5;
static const float PRESS_FULL = 2200;   // raw counts above the resting value that count as "full" pressure
static const float PRESS_MIN = 0.06;    // below this the card counts as not held
static const float CHARGE_UP = 0.12;    // charge per second at full pressure
static const float CHARGE_DOWN = 0.05;  // drain per second when not held
static const float BPM_MIN = 55, BPM_MAX = 140;
static const int PWM_MIN = 110, PWM_MAX = 200;   // 3 V motor on 5 V: stay below full duty
static const int LUB_MS = 90, GAP_MS = 130, DUB_MS = 70;

float base = 0, pressure = 0, charge = 0;
uint32_t lastBeat = 0, lastLog = 0, lastTick = 0;

void setup() {
  Serial.begin(115200);
  pinMode(PIN_MOTOR, OUTPUT);
  analogWrite(PIN_MOTOR, 0);
  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);          // full 0..3.3 V range
  delay(300);
  long sum = 0;
  for (int i = 0; i < 100; i++) { sum += analogRead(PIN_FSR); delay(10); }
  base = sum / 100.0f;
  Serial.printf("postcard v1 ready, resting value %.0f\n", base);
  lastTick = millis();
}

void loop() {
  uint32_t now = millis();
  float dt = (now - lastTick) / 1000.0f;
  lastTick = now;

  // pressure 0..1, lightly smoothed
  int raw = analogRead(PIN_FSR);
  float p = constrain((raw - base) / PRESS_FULL, 0.0f, 1.0f);
  pressure += (p - pressure) * 0.3f;
  bool held = pressure > PRESS_MIN;

  // charge: rises while held (faster when pressed harder), drains slowly after letting go
  if (held) charge = min(1.0f, charge + CHARGE_UP * pressure * dt * 2.0f);
  else charge = max(0.0f, charge - CHARGE_DOWN * dt);

  // heartbeat: tempo from the charge, strength from the pressure (a faint beat while it still carries charge)
  float bpm = BPM_MIN + (BPM_MAX - BPM_MIN) * charge;
  bool beating = held || charge > 0.02f;
  float strength = held ? pressure : 0.25f * charge;
  int pwm = beating ? (int)(PWM_MIN + (PWM_MAX - PWM_MIN) * constrain(strength, 0.0f, 1.0f)) : 0;

  uint32_t interval = (uint32_t)(60000.0f / bpm);
  if (now - lastBeat >= interval) lastBeat = now;
  uint32_t t = now - lastBeat;
  int out = 0;
  if (beating) {
    if (t < LUB_MS) out = pwm;                                          // lub
    else if (t >= LUB_MS + GAP_MS && t < LUB_MS + GAP_MS + DUB_MS)
      out = (int)(pwm * 0.65f);                                        // dub (softer)
  }
  analogWrite(PIN_MOTOR, out);

  if (now - lastLog >= 100) {
    lastLog = now;
    Serial.printf("raw %4d  pressure %.2f  charge %.2f  bpm %3.0f  %s\n", raw, pressure, charge, bpm,
                  held ? "held" : (beating ? "fading" : "still"));
  }
  delay(5);
}
